function Assert-ExactFields($Value, [string[]]$Names) {
    $actual = @($Value.PSObject.Properties.Name | Sort-Object)
    $expected = @($Names | Sort-Object)
    if (($actual -join "`n") -cne ($expected -join "`n")) { throw 'Unexpected record fields' }
}
function Get-CanonicalPath([string]$Path) {
    if ([string]::IsNullOrWhiteSpace($Path) -or -not [IO.Path]::IsPathRooted($Path) -or
        $Path.Contains([char]0) -or $Path -match '(^|[\\/])\.\.([\\/]|$)') { throw 'Absolute non-traversing path required' }
    $full = [IO.Path]::GetFullPath($Path)
    if ($full.Substring([IO.Path]::GetPathRoot($full).Length).Contains(':')) { throw 'Alternate data stream forbidden' }
    $cursor = $full
    while ($cursor) {
        if ([IO.File]::Exists($cursor) -or [IO.Directory]::Exists($cursor)) {
            if (([IO.File]::GetAttributes($cursor) -band [IO.FileAttributes]::ReparsePoint) -ne 0) { throw 'Reparse path forbidden' }
        }
        $parent = [IO.Path]::GetDirectoryName($cursor)
        if ($parent -eq $cursor) { break }
        $cursor = $parent
    }
    return $full
}
function Get-StableIdentity([string]$Path) {
    $full = Get-CanonicalPath $Path
    $before = Get-Item -LiteralPath $full -Force -ErrorAction Stop
    if ($before.PSIsContainer) { throw 'Ordinary file required' }
    $length = $before.Length; $ticks = $before.LastWriteTimeUtc.Ticks
    $stream=[IO.File]::Open($full,[IO.FileMode]::Open,[IO.FileAccess]::Read,[IO.FileShare]::Read)
    $sha=[Security.Cryptography.SHA256]::Create()
    try { $hash=([BitConverter]::ToString($sha.ComputeHash($stream))).Replace('-','').ToLowerInvariant() }
    finally { $sha.Dispose(); $stream.Dispose() }
    $after = Get-Item -LiteralPath $full -Force -ErrorAction Stop
    if ($length -ne $after.Length -or $ticks -ne $after.LastWriteTimeUtc.Ticks) { throw 'File changed during hash' }
    return [pscustomobject]@{path=$full; size_bytes=$length; sha256=$hash}
}
function Assert-Identity($Spec) {
    if ($Spec.sha256 -isnot [string] -or $Spec.sha256 -cnotmatch '^[0-9a-f]{64}$' -or
        ($Spec.size_bytes -isnot [int] -and $Spec.size_bytes -isnot [long]) -or $Spec.size_bytes -lt 0) { throw 'Invalid identity' }
    $actual = Get-StableIdentity $Spec.path
    if ($actual.sha256 -cne $Spec.sha256 -or $actual.size_bytes -ne $Spec.size_bytes) { throw ('Identity mismatch: ' + $Spec.path) }
}
function Read-PinnedJson($Spec, [long]$Limit) {
    Assert-Identity $Spec
    if ($Spec.size_bytes -gt $Limit) { throw 'JSON byte limit' }
    $stream=[IO.File]::Open((Get-CanonicalPath $Spec.path),[IO.FileMode]::Open,[IO.FileAccess]::Read,[IO.FileShare]::Read)
    $buffer=New-Object byte[] ([int]$Spec.size_bytes)
    try {
        $offset=0
        while ($offset -lt $buffer.Length) {
            $count=$stream.Read($buffer,$offset,$buffer.Length-$offset)
            if ($count -eq 0) { throw 'JSON shortened during read' }; $offset+=$count
        }
        if ($stream.ReadByte() -ne -1) { throw 'JSON grew during read' }
    } finally { $stream.Dispose() }
    $sha=[Security.Cryptography.SHA256]::Create()
    try { $hash=([BitConverter]::ToString($sha.ComputeHash($buffer))).Replace('-','').ToLowerInvariant() }
    finally { $sha.Dispose() }
    if ($hash -cne $Spec.sha256) { throw 'JSON bytes changed after hash' }
    return ([Text.Encoding]::UTF8.GetString($buffer) | ConvertFrom-Json)
}
function Write-NewJson([string]$Path, $Value) {
    $full = Get-CanonicalPath $Path
    $raw = [Text.Encoding]::UTF8.GetBytes(($Value | ConvertTo-Json -Depth 25 -Compress))
    $stream = [IO.File]::Open($full, [IO.FileMode]::CreateNew, [IO.FileAccess]::Write, [IO.FileShare]::None)
    try { $stream.Write($raw,0,$raw.Length); $stream.Flush($true) } finally { $stream.Dispose() }
}
function Assert-MemberManifests($Manifests) {
    $seen = @{}; $total = 0
    if (@($Manifests).Count -gt 64) { throw 'Too many manifests' }
    foreach ($manifest in $Manifests) {
        Assert-ExactFields $manifest @('path','sha256','size_bytes','members')
        Assert-Identity $manifest
        if ($manifest.size_bytes -gt 16777216) { throw 'Manifest byte limit' }
        Assert-ExactFields $manifest.members @('roots','count','complete_roots')
        if ($manifest.members.count -le 0 -or $manifest.members.count -gt 50000) { throw 'Manifest count limit' }
        $total += $manifest.members.count
        if ($total -gt 100000) { throw 'Total member limit' }
        $rows = @(Read-PinnedJson $manifest 16777216)
        if ($rows.Count -ne $manifest.members.count) { throw 'Manifest count mismatch' }
        $local = @{}; $roots = @($manifest.members.roots | ForEach-Object { Get-CanonicalPath $_ })
        if ($roots.Count -eq 0 -or $roots.Count -gt 32) { throw 'Invalid roots' }
        foreach ($row in $rows) {
            $names=@($row.PSObject.Properties.Name)
            if ($names -contains 'mtime_ns') { Assert-ExactFields $row @('path','size_bytes','sha256','mtime_ns') }
            else { Assert-ExactFields $row @('path','size_bytes','sha256') }
            $path = Get-CanonicalPath $row.path
            $contained = $false
            foreach ($root in $roots) {
                if ($path.StartsWith($root.TrimEnd('\')+'\',[StringComparison]::OrdinalIgnoreCase)) { $contained=$true }
            }
            if (-not $contained -or $local.ContainsKey($path)) { throw 'Outside root or duplicate member' }
            $local[$path] = $true
            $identity = [string]$row.size_bytes + ':' + $row.sha256
            if ($seen.ContainsKey($path) -and $seen[$path] -cne $identity) { throw 'Conflicting member identity' }
            Assert-Identity $row
            $seen[$path] = $identity
        }
        foreach ($root in @($manifest.members.complete_roots)) {
            $root = Get-CanonicalPath $root
            if ($roots -notcontains $root) { throw 'Unreviewed complete root' }
            $actual = @{}
            $pending=[Collections.Generic.Stack[string]]::new(); $pending.Push($root)
            while ($pending.Count -gt 0) {
                foreach ($item in @(Get-ChildItem -LiteralPath $pending.Pop() -Force -ErrorAction Stop)) {
                    $path = Get-CanonicalPath $item.FullName
                    if ($item.PSIsContainer) { $pending.Push($path) } else { $actual[$path]=$true }
                }
            }
            $expected = @($local.Keys | Where-Object { $_.StartsWith($root.TrimEnd('\')+'\',[StringComparison]::OrdinalIgnoreCase) })
            if ($actual.Count -ne $expected.Count) { throw 'Complete tree membership changed' }
            foreach ($path in $expected) { if (-not $actual.ContainsKey($path)) { throw 'Complete tree member missing' } }
        }
    }
}
function Assert-LaunchCoverage($Record, $Packet) {
    $expectedMembers=@{}; $expectedDirect=@{}
    foreach ($property in $Packet.declared_files.PSObject.Properties) {
        $spec=$property.Value; $path=Get-CanonicalPath $spec.path
        if ($null -ne $spec.members) { $expectedMembers[$path]=$spec }
        else { $expectedDirect[$path]=$spec }
    }
    if ($expectedMembers.Count -eq 0 -or $expectedDirect.Count -eq 0) { throw 'Empty packet coverage' }
    $actualMembers=@{}
    foreach ($spec in @($Record.member_manifests)) {
        $path=Get-CanonicalPath $spec.path
        if ($actualMembers.ContainsKey($path) -or -not $expectedMembers.ContainsKey($path)) { throw 'Unexpected/duplicate outer manifest' }
        $expected=$expectedMembers[$path]
        if ($spec.sha256 -cne $expected.sha256 -or $spec.size_bytes -ne $expected.size_bytes -or
            $spec.members.count -ne $expected.members.count -or
            ((@($spec.members.roots | Sort-Object) -join "`n") -cne (@($expected.members.roots | Sort-Object) -join "`n")) -or
            ((@($spec.members.complete_roots | Sort-Object) -join "`n") -cne (@($expected.members.complete_roots | Sort-Object) -join "`n"))) { throw 'Outer manifest differs from packet' }
        $actualMembers[$path]=$true
    }
    if ($actualMembers.Count -ne $expectedMembers.Count) { throw 'Incomplete outer manifest coverage' }
    $actualDirect=@{}
    foreach ($spec in @($Record.bootstrap_files)) {
        Assert-ExactFields $spec @('path','size_bytes','sha256')
        $path=Get-CanonicalPath $spec.path
        if ($actualDirect.ContainsKey($path) -or -not $expectedDirect.ContainsKey($path)) { throw 'Unexpected/duplicate bootstrap file' }
        $expected=$expectedDirect[$path]
        if ($spec.sha256 -cne $expected.sha256 -or $spec.size_bytes -ne $expected.size_bytes) { throw 'Bootstrap identity differs from packet' }
        $actualDirect[$path]=$true
    }
    if ($actualDirect.Count -ne $expectedDirect.Count) { throw 'Incomplete bootstrap coverage' }
}
